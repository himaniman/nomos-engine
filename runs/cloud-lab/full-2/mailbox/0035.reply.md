HYPOTHESIS: Axis: an internal TEMPORAL state. Area-restricted search failed in SPACE (#12 neutral; #17 31.4, its patch field suppressed the course wherever a meal had been, long after the clump was empty). Here the same idea runs in TIME: a hunger clock kept at the agent's own cell (reset by a meal, filling over ~300 ticks, gathered back after each scroll like the course) is broadcast over the map and scales the course's weight on frontier goals — right after eating the frontier is weighed evenly (comb the clump: stocked rooms and trees hold two morsels), and the long course takes the walk over again as the clock fills, wherever the agent then is. Built on #21 (unit course + doors); the clock is the one change.

channel thing
channel solid
channel free
channel here
channel bad
channel leak
channel value

channel confinement
channel unk
channel room
channel front
channel dwell: init 0
channel crowd: init 0
channel thinwall
channel recent: init 0
channel coursex: init 0
channel coursey: init 0
channel unitx
channel unity
channel beamxp: init 0
channel beamxn: init 0
channel beamyp: init 0
channel beamyn: init 0
channel ahead
channel hidden
channel door
channel injection
channel feast
channel fasting: init 0
channel appetite: init 0

  when 1
  set recent = max(here, 0.95 * recent)
rule course_x:
  when 1
  set coursex = here * (0.998 * (coursex + sum8(coursex)) + 0.002 * (shift(recent, 1, 0) - shift(recent, -1, 0)))
rule course_y:
  when 1
  set coursey = here * (0.998 * (coursey + sum8(coursey)) + 0.002 * (shift(recent, 0, 1) - shift(recent, 0, -1)))
# RE-AIM: the course as a UNIT direction at the agent's cell. A looping walk averages to a short
# vector whose pull used to vanish; normalised, it keeps full strength in whatever way the loop
# drifts, so an orbit is broken instead of tolerated
rule unit_x:
  when 1
  set unitx = here * coursex / (sqrt(coursex * coursex + coursey * coursey) + 0.02)
rule unit_y:
  when 1
  set unity = here * coursey / (sqrt(coursex * coursex + coursey * coursey) + 0.02)
rule beam_xp:
  when 1
  set beamxp = where(here, max(unitx, 0), 0.97 * max8(beamxp))
rule beam_xn:
  when 1
  set beamxn = where(here, max(0 - unitx, 0), 0.97 * max8(beamxn))
rule beam_yp:
  when 1
  set beamyp = where(here, max(unity, 0), 0.97 * max8(beamyp))
rule beam_yn:
  when 1
  set beamyn = where(here, max(0 - unity, 0), 0.97 * max8(beamyn))
rule along_course:
  when 1
  set ahead = ((posx() - 0.4828) * (beamxp - beamxn) + (posy() - 0.4828) * (beamyp - beamyn)) / (sqrt((posx() - 0.4828) * (posx() - 0.4828) + (posy() - 0.4828) * (posy() - 0.4828)) + 0.01)
rule dwell_time:
  when 1
  set dwell = min(1, 0.998 * dwell + 0.02 * here)
rule frustration:
  when 1
  set crowd = 0.5 * dwell + 0.5 * avg8(crowd)
rule unknown:
  when 1
  set unk = (free < 0.2) * (solid < 0.5)
rule roominess:
  when 1
  set room = 0.5 * unk + 0.5 * avg8(room)
rule frontier:
  when 1
  set front = unk * (max8(free) > 0.5)
rule thin_walls:
  when 1
  set thinwall = solid * max((1 - shift(solid, 1, 0)) * (1 - shift(solid, -1, 0)), (1 - shift(solid, 0, 1)) * (1 - shift(solid, 0, -1)))
# an unexplored space BEHIND a wall: unknown cells walled off from every known-free cell
rule walled_off:
  when 1
  set hidden = unk * (max8(free) < 0.5) * (max8(solid) > 0.5)
# DOORS AS FRONTIER: a thin wall whose hue has been taught to give (leak above the innate 0.5),
# with known ground on one side and hidden space on the other, is a way into a new room
rule doorways:
  when 1
  set door = thinwall * clamp(2 * (leak - 0.5), 0, 1) * (max8(free) > 0.5) * max8(hidden)
rule linger:
  when 1
  set ghost = max(source, thing, 0.99 * ghost) * (1 - here) * (1 - max8(here))
rule calc_conf:
  when 1
  set confinement = sum8(solid)
# A CLOCK OF HUNGER kept at the agent's cell (gathered back after each scroll, like the course):
# reset by a meal, it fills over ~300 ticks; broadcast over the map it says how long since food
rule meal:
  when 1
  set feast = here * max(thing, max8(thing)) * (1 - bad)
rule fasting_clock:
  when 1
  set fasting = here * where(feast > 0.5, 0, min(1, fasting + sum8(fasting) + 0.0015))
rule appetite_field:
  when 1
  set appetite = where(here, fasting, 0.97 * max8(appetite))
# TEMPORAL intermittent search: just after a meal the course has no say (search where food was,
# clumps hold more), and it takes the walk over again as hunger grows
rule inject:
  when 1
  set injection = max(ghost, 0.3 * front * room * room * clamp(1 + ahead * appetite, 0, 2), 0.4 * door, 0.04 * unk) / (1 + 0.5 * confinement * (1 - crowd)) * (1 - bad)
rule wave:
  when 1
  set value = max(injection, max(0.94 * max4(value), 0.916214 * max8(value))) * (1 - solid * (1 - min(1, leak * (0.3 + 0.7 * thinwall) * (1 + 3 * crowd))))
